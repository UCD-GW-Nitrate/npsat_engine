cmake_minimum_required(VERSION 3.1)
find_package(deal.II 9.3 REQUIRED HINTS ${DEAL_II_DIR} $ENV{DEAL_II_DIR})
deal_ii_initialize_cached_variables()
project(interpolation_tests CXX)
add_executable(interpolation_tests interpolation_tests.cc)
set_target_properties(interpolation_tests PROPERTIES
  CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
deal_ii_setup_target(interpolation_tests)
enable_testing()
add_test(NAME interpolation_serial COMMAND interpolation_tests ${CMAKE_CURRENT_SOURCE_DIR}/../examples/interpolation_v2)
add_executable(solver_callback_tests solver_callback_tests.cc)
set_target_properties(solver_callback_tests PROPERTIES
  CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
deal_ii_setup_target(solver_callback_tests)
add_test(NAME solver_callback COMMAND solver_callback_tests)
add_executable(shared_reader_probe shared_reader_probe.cc)
target_include_directories(shared_reader_probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../myheaders)
set_target_properties(shared_reader_probe PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
deal_ii_setup_target(shared_reader_probe)
add_test(NAME shared_reader COMMAND shared_reader_probe ${CMAKE_CURRENT_SOURCE_DIR}/../examples/shared_flow/input)
set(NPSAT_V2_REFERENCE "" CACHE PATH "Optional trace_dbg npsat_flow directory for reader comparison")
if(NPSAT_V2_REFERENCE)
  add_executable(v2_reader_probe shared_reader_probe.cc)
  target_include_directories(v2_reader_probe PRIVATE ${NPSAT_V2_REFERENCE})
  target_compile_definitions(v2_reader_probe PRIVATE NPSAT_V2_REFERENCE)
  set_target_properties(v2_reader_probe PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  deal_ii_setup_target(v2_reader_probe)
endif()
